38 void analyzePDBAtoms(const FileName &fn_pdb, const std::string &typeOfAtom, int &numberOfAtoms, pdbInfo &at_pos)
39 {
40  //Open the pdb file
41  std::ifstream f2parse;
42  f2parse.open(fn_pdb.c_str());
43 
44  numberOfAtoms = 0;
45 
46  while (!f2parse.eof())
47  {
48  std::string line;
49  getline(f2parse, line);
50 
51  // The type of record (line) is defined in the first 6 characters of the pdb
52  std::string typeOfline = line.substr(0,4);
53 
54  if ( (typeOfline == "ATOM") || (typeOfline == "HETA"))
55  {
56  // Type of Atom
57  std::string at;
58  try
59  {
60  at = line.substr(13,2);
61  }catch (const std::out_of_range& oor)
62  {
63  std::cerr << "Out of Range error: One of the pdb lines failed selecting the atom type" << '\n';
64  }
65 
66  if (at == typeOfAtom)
67  {
68  // Atom positions
69  numberOfAtoms++;
70  double x = textToFloat(line.substr(30,8));
71  double y = textToFloat(line.substr(38,8));
72  double z = textToFloat(line.substr(46,8));
73 
74  // storing coordinates
75  at_pos.x.push_back(x);
76  at_pos.y.push_back(y);
77  at_pos.z.push_back(z);
78 
79  // Residue Number
80  int resi = (int) textToFloat(line.substr(23,5));
81  at_pos.residue.push_back(resi);
82 
83  // Getting the bfactor = 8pi^2*u
84  double bfactorRad = sqrt(textToFloat(line.substr(60,6))/(8*PI*PI));
85  at_pos.b.push_back(bfactorRad);
86 
87  // Covalent radius of the atom
88  double rad = atomCovalentRadius(line.substr(13,2));
89  at_pos.atomCovRad.push_back(rad);
90  }
91  }
92  }
93 }

233 /* Compute geometry -------------------------------------------------------- */
234 void computePDBgeometry(const std::string &fnPDB,
235  Matrix1D<double> &centerOfMass,
236  Matrix1D<double> &limit0, Matrix1D<double> &limitF,
237  const std::string &intensityColumn)
238 {
239  // Initialization
240  centerOfMass.initZeros(3);
241  limit0.resizeNoCopy(3);
242  limitF.resizeNoCopy(3);
243  limit0.initConstant(1e30);
244  limitF.initConstant(-1e30);
245  double total_mass = 0;
246 
247  // Open the file
248  std::ifstream fh_pdb;
249  fh_pdb.open(fnPDB.c_str());
250  if (!fh_pdb)
251  REPORT_ERROR(ERR_IO_NOTEXIST, fnPDB);
252 
253  // Process all lines of the file
254  int col=1;
255  if (intensityColumn=="Bfactor")
256  col=2;
257  while (!fh_pdb.eof())
258  {
259  // Read a ATOM line
260  std::string line;
261  getline(fh_pdb, line);
262  if (line == "")
263  continue;
264  std::string kind = line.substr(0,4);
265  if (kind != "ATOM" && kind!="HETA")
266  continue;
267 
268  // Extract atom type and position
269  // Typical line:
270  // ATOM 909 CA ALA A 161 58.775 31.984 111.803 1.00 34.78
271  std::string atom_type = line.substr(13,2);
272  double x = textToFloat(line.substr(30,8));
273  double y = textToFloat(line.substr(38,8));
274  double z = textToFloat(line.substr(46,8));
275 
276  // Update center of mass and limits
277  if (x < XX(limit0))
278  XX(limit0) = x;
279  else if (x > XX(limitF))
280  XX(limitF) = x;
281  if (y < YY(limit0))
282  YY(limit0) = y;
283  else if (y > YY(limitF))
284  YY(limitF) = y;
285  if (z < ZZ(limit0))
286  ZZ(limit0) = z;
287  else if (z > ZZ(limitF))
288  ZZ(limitF) = z;
289  double weight;
290  if (atom_type=="EN")
291  {
292  if (col==1)
293  weight=textToFloat(line.substr(54,6));
294  else if (col==2)
295  weight=textToFloat(line.substr(60,6));
296  }
297  else
298  {
299  if (kind=="HETA")
300  continue;
301  weight=(double) atomCharge(atom_type);
302  }
303  total_mass += weight;
304  XX(centerOfMass) += weight * x;
305  YY(centerOfMass) += weight * y;
306  ZZ(centerOfMass) += weight * z;
307  }
308 
309  // Finish calculations
310  centerOfMass /= total_mass;
311 
312  // Close file
313  fh_pdb.close();
314 }

318  const Matrix2D<double> &A, bool centerPDB,

695 /* Electron form factor in Fourier ----------------------------------------- */
696 double electronFormFactorFourier(double f, const Matrix1D<double> &descriptors)
697 {
698  double retval=0;
699  for (int i=1; i<=5; i++)
700  {
701  double ai=VEC_ELEM(descriptors,i);
702  double bi=VEC_ELEM(descriptors,i+5);
703  retval+=ai*exp(-bi*f*f);
704  }
705  return retval;
706 }

708 /* Electron form factor in real space -------------------------------------- */
709 /* We know the transform pair
710  sqrt(pi/b)*exp(-x^2/(4*b)) <----> exp(-b*W^2)
711 
712  We also know that
713 
714  X(f)=sum_i ai exp(-bi*f^2)
715 
716  Therefore, using W=2*pi*f
717 
718  X(W)=sum_i ai exp(-bi/(2*pi)^2*W^2)
719 
720  Thus, the actual b for the inverse Fourier transform is bi/(2*pi)^2
721  And we have to divide by 2*pi to account for the Jacobian of the
722  transformation.
723 */
724 double electronFormFactorRealSpace(double r,
725  const Matrix1D<double> &descriptors)
726 {
727  double retval=0;
728  for (int i=1; i<=5; i++)
729  {
730  double ai=descriptors(i);
731  double bi=descriptors(i+5);
732  double b=bi/(4*PI*PI);
733  retval+=ai*sqrt(PI/b)*exp(-r*r/(4*b));
734  }
735  retval/=2*PI;
736  return retval;
737 }

771 void fhlpf(const MultidimArray<double> &f, const MultidimArray<double> &filter,
772  int M, MultidimArray<double> &convolution)
773 {

780  auxFilter.selfWindow(STARTINGX(filter)-Nmax,FINISHINGX(filter)+Nmax);

785  FourierTransform(auxFilter,Filter);

789  // amplitude factor
